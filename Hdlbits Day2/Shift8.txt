module top_module ( 
    input clk, 
    input [7:0] d, 
    input [1:0] sel, 
    output [7:0] q 
);
    wire [7:0] q_reg1,q_reg2,q_reg3;
    my_dff8 d1(
        .clk(clk),
        .d(d),
        .q(q_reg1)
    );
    my_dff8 d2(
        .clk(clk),
        .d(q_reg1),
        .q(q_reg2)
    );
    my_dff8 d3(
        .clk(clk),
        .d(q_reg2),
        .q(q_reg3)
    );
    
    assign q= (sel=='b00) ? d:
        (sel=='b01) ? q_reg1:
        (sel=='b10) ? q_reg2: q_reg3;
endmodule
